```gdscript
func _find_and_highlight_row(node: AssemblyNode, target_path: String) -> bool:
    # 精确匹配则遍历 _tree_container 找对应行并高亮
    if node_path == target_path:
        for child in _tree_container.get_children():
            if child.get_meta("node_ref").get_path_string() == target_path:
                _set_row_highlight(child, SELECTED_BG_COLOR)
                return true
        return false
    # 递归检查 4 个槽位（MOUNT/MUZZLE/MAGAZINE/BULLET）
    for slot_type in AssemblyNode.SlotType.values():
        child = node.slots[slot_type]
        if child != null and _find_and_highlight_row(child, target_path):
            return true
    return false
```
